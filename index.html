Kostecki", } race_state = {} def parse_xml_packet(raw_xml: str): """Parses live Natsoft XML delta packets into clean leaderboard state.""" try: if not raw_xml.startswith("<"): return try: root = ET.fromstring(raw_xml) except ET.ParseError: root = ET.fromstring(f"{raw_xml}") # Extract car records p_nodes = root.findall(".//P") for p in p_nodes: raw_c = p.attrib.get("C") if not raw_c: continue car_num = str(raw_c).strip() existing = race_state.get(car_num, {}) # Parse position strictly pos_val = p.attrib.get("P") or p.attrib.get("LP") try: pos_int = int(pos_val) if pos_val else existing.get("pos", 99) except ValueError: pos_int = existing.get("pos", 99) # In Natsoft XML, 'LL' is Completed Laps; 'L' on
